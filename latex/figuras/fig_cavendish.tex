  \draw[azulprof, line width=0.8pt] (0,0) circle (\Re);
  \draw[azulprof, line width=0.8pt] (-\Re,0) arc (180:360:{\Re} and 0.45);
  \draw[azulprof, line width=0.5pt, densely dashed] (-\Re,0) arc (180:0:{\Re} and 0.45);
  \shade[ball color=azulacero!70, opacity=0.95] (0,0) circle (\Ri);

  \draw[grispizarra, line width=0.3pt] (2.7,0.9) -- (1.3,0.9); \node[font=\footnotesize, anchor=west] at (2.75,0.9) {hemisferios};
  \draw[grispizarra, line width=0.3pt] (2.7,-0.9) -- (1.5,-0.9); \node[font=\footnotesize, anchor=west] at (2.75,-0.9) {cáscara};
  \draw[grispizarra, line width=0.3pt] (2.7,0.0) -- (0.62,0.0); \node[font=\footnotesize, anchor=west] at (2.75,0.0) {esfera interior};
  \end{tikzpicture}\\[2pt]{\small\scshape (a)}
\end{minipage}\hfill
\begin{minipage}[b]{0.52\textwidth}\centering
\begin{tikzpicture}[x=0.84cm, y=0.84cm]
  % (b) cono de vértice P interior a la cáscara (radio 2.2, centro C)
  \def\Rr{2.2}
  \coordinate (C) at (0,0);
  \coordinate (P) at (0.3,-0.95);
  \def\dd{20}\def\ap{8}
  \draw[azulprof, line width=0.9pt, name path=circ, fill=azulfondo] (C) circle (\Rr);
  \foreach \dir/\nm in {\dd-\ap/a,\dd+\ap/b,\dd/m}{
    \path[name path=ray\nm] (P) -- ++({\dir}:5);
    \path[name path=rev\nm] (P) -- ++({\dir+180}:5);
    \path[name intersections={of=circ and ray\nm, by=A\nm}];
    \path[name intersections={of=circ and rev\nm, by=B\nm}];}
  \fill[azulclaro!80, opacity=0.55] (P) -- (Aa) -- (Ab) -- cycle;
  \fill[azulclaro!80, opacity=0.55] (P) -- (Ba) -- (Bb) -- cycle;
  \draw[azulacero, line width=0.45pt] (Aa) -- (Ba) (Ab) -- (Bb);
  \draw[azulprof, line width=0.7pt] (Bm) -- (Am);
  \draw[verdeteo, line width=0.55pt, densely dashed] (C) -- (Am) (C) -- (Bm);
  \draw[acento, line width=3pt] (Aa) -- (Ab);
  \draw[acento, line width=3pt] (Ba) -- (Bb);
  \fill[azulprof] (C) circle (1.4pt); \node[font=\small, anchor=south] at ($(C)+(0,0.06)$) {$C$};
  \fill[azulprof] (P) circle (1.6pt); \node[font=\small, anchor=south east] at ($(P)+(-0.02,0.08)$) {$P$};
  % distancias r1, r2 acotadas por debajo del cono
  \draw[azulprof, <->, line width=0.45pt] ($(P)!0.12!(Am)+(\dd-90:0.42)$) -- ($(Am)+(\dd-90:0.42)$) node[midway, below right=-2pt, font=\scriptsize] {$r_2$};
  \draw[azulprof, <->, line width=0.45pt] ($(Bm)+(\dd-90:0.42)$) -- ($(P)!0.12!(Bm)+(\dd-90:0.42)$) node[midway, below right=-2pt, font=\scriptsize] {$r_1$};
  \node[font=\small, text=acento, anchor=west] at ($(Am)+(0.15,0.1)$) {$\dif S_2$};
  \node[font=\small, text=acento, anchor=east] at ($(Bm)+(-0.15,-0.1)$) {$\dif S_1$};
  % ángulos alpha
  \draw[verdeteo, line width=0.6pt] let \p1=(Am), \n1={atan2(\y1,\x1)+180} in
     (Am) ++(\n1:0.75) arc[start angle=\n1, end angle=\dd+180, radius=0.75];
  \draw let \p1=(Am), \n1={atan2(\y1,\x1)+180} in
     node[font=\scriptsize, text=verdeteo] at ($(Am)+({(\n1+\dd+180)/2}:1.02)$) {$\alpha$};
  \draw[verdeteo, line width=0.6pt] let \p1=(Bm), \n1={atan2(\y1,\x1)+180} in
     (Bm) ++(\n1:0.75) arc[start angle=\n1, end angle=\dd, radius=0.75];
  \draw let \p1=(Bm), \n1={atan2(\y1,\x1)+180} in
     node[font=\scriptsize, text=verdeteo] at ($(Bm)+({(\n1+\dd)/2}:1.02)$) {$\alpha$};
\end{tikzpicture}\\[2pt]{\small\scshape (b)}
\end{minipage}
\caption{(a)~Experimento de Cavendish: una esfera conductora interior se conecta, mediante un hilo, a una cáscara formada por dos hemisferios, y se cargan; al retirar el hilo y separar los hemisferios,
la esfera interior resulta descargada. (b)~Argumento geométrico: un cono de ángulo sólido $\dOm$ con vértice en un punto interior $P$ recorta dos elementos, $\dif S_1$ y $\dif S_2$,
a distancias $r_1$ y $r_2$ de $P$; ambos forman con la cuerda el mismo ángulo $\alpha$ (el triángulo formado por los dos radios y la cuerda es isósceles).}
\label{fig:cavendish}
