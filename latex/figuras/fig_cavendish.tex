  % (b) cono de ángulo sólido que corta a la cáscara en dos elementos
  \def\Rr{1.75}
  \draw[azulprof, line width=0.8pt, name path=circ] (0,0) circle (\Rr);
  \coordinate (P) at (0.55,0.30);

    \path[name path=ray\nm] (P) -- ++(\dir:4);
    \path[name path=rev\nm] (P) -- ++({\dir+180}:4);
    \path[name intersections={of=circ and ray\nm, by=A\nm}];
    \path[name intersections={of=circ and rev\nm, by=B\nm}];
    \draw[azulacero, line width=0.5pt] (B\nm) -- (A\nm);}

\caption{(a)~Experimento de Cavendish: una esfera conductora interior se conecta, mediante un hilo,
a una cáscara formada por dos hemisferios, y se cargan; al retirar el hilo y separar los hemisferios,
la esfera interior resulta descargada. (b)~Argumento geométrico: un cono de ángulo sólido
$\dOm$ con vértice en un punto interior $P$ recorta dos elementos, $\dif S_1$ y $\dif S_2$, a
distancias $r_1$ y $r_2$ de $P$; ambos forman con la cuerda el mismo ángulo $\alpha$ (el triángulo
formado por los radios y la cuerda es isósceles).}
\label{fig:cavendish}
