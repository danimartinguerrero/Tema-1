  \node[font=\small, text=azulprof, anchor=west] at ($(S)!0.45!(P)+(0.0,0.08,0)$) {$\vc{R}$};
  \draw[->, azulacero, line width=1.1pt] (0,0,0) -- (P);
  \node[font=\small, text=azulprof, anchor=east] at (0,0,2.9) {$\vc{r}=z\,\uz$};
